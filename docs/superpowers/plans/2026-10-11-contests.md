# Contests Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Codeforces-style contests: admin-created contests with attached unpublished problems that auto-publish when a public contest ends, registration, contest-window submissions, and full standings with points/penalty/freeze.

**Architecture:** Layered as the existing codebase (api → service → repository; entities in `common`). New migration V6 extends the existing V1 contest tables. A `@Scheduled` publisher flips `is_published` on public-contest problems at contest end (first scheduled task in the repo — adds `@EnableScheduling`). Contest problem statements are served by contest-scoped endpoints; `SubmissionCreator` gains one contest-aware validation branch. Standings are aggregated in Java from `submissions` with a 15s Redis cache-aside.

**Tech Stack:** Java 21 · Spring Boot 3.x · Spring Data JPA · Flyway · PostgreSQL · Redis (StringRedisTemplate) · React 18 + TypeScript + Vite · React Query · JUnit 5 + Mockito + Testcontainers · Vitest + Testing Library.

**Spec:** `docs/superpowers/specs/2026-10-11-contests-design.md` — read it before starting; this plan implements it task by task.

## Global Constraints

- Layering per AGENTS.md: controllers thin; `@Transactional` on service methods; entities in `common` with no web/AMQP deps; DTOs are immutable records; constructor injection only; admin authorization at BOTH controller (`@PreAuthorize`) and service layer.
- Every schema change is migration `V6__contests.sql` (never edit applied migrations). Every index justified in the migration comment (PRD §8).
- Error mapping conventions (existing): `ResourceNotFoundException`→404, `BusinessRuleException`→400, `DuplicateResourceException`→409, `ResourceInUseException`→409; new `ContestRegistrationRequiredException`→403 (added to `GlobalExceptionHandler`).
- No new dependencies (Maven or npm). No changes to `execution-worker`.
- DoD per task: `mvn -B verify` (backend tasks; Docker required for tagged integration tests) and/or `npm run lint && npm run test && npm run build` (frontend tasks), exactly as each task lists.
- Commit steps are task boundaries with the exact Conventional Commit messages given. Only commit if the human has approved committing (they have, for this session, on `dev`); never push; never stage `docs/superpowers/plans/2026-10-07-phase12-admin-dashboard.md` or the untracked docs/superpowers spec/plan files.
- Status values are derived, never stored: `ContestStatus.of(start, end, now)`. All times `Instant`/UTC; API serializes ISO-8601.
- Do not rename existing endpoints/fields; only the edits this plan lists.

```sql
-- Contests (PRD §7.8): visibility, freeze, and auto-publish marker on the V1 contest
-- tables. Indexes justified by query pattern (PRD §8):
--   idx_contests_times           -> contest listing ordered by time
--   uq_contest_problems_problem  -> one contest per problem AND submission-validation
--                                   lookup by problem_id (unique index covers both)
--   idx_submissions_problem_time -> standings aggregation per problem window
ALTER TABLE contests
  ADD COLUMN is_private BOOLEAN NOT NULL DEFAULT FALSE,
  ADD COLUMN freeze_minutes INT NOT NULL DEFAULT 60,
  ADD COLUMN problems_published_at TIMESTAMPTZ;

ALTER TABLE contests
  ADD CONSTRAINT chk_contests_times CHECK (end_time > start_time);
ALTER TABLE contests
  ADD CONSTRAINT chk_contests_freeze CHECK (freeze_minutes >= 0);

CREATE INDEX idx_contests_times ON contests (start_time, end_time);
CREATE UNIQUE INDEX uq_contest_problems_problem ON contest_problems (problem_id);
CREATE INDEX idx_submissions_problem_time ON submissions (problem_id, submitted_at);
```

```java
package com.onlinejudge.common.enums;

import java.time.Instant;

/** Contest lifecycle derived from its time window — never stored (PRD §7.8). */
public enum ContestStatus {
    UPCOMING,
    RUNNING,
    ENDED;

    public static ContestStatus of(Instant start, Instant end, Instant now) {
        if (now.isBefore(start)) {
            return UPCOMING;
        }
        return now.isBefore(end) ? RUNNING : ENDED;
    }
}
```

```java
package com.onlinejudge.common.entity;

import java.time.Instant;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.FetchType;
import jakarta.persistence.GeneratedValue;
import jakarta.persistence.GenerationType;
import jakarta.persistence.Id;
import jakarta.persistence.JoinColumn;
import jakarta.persistence.ManyToOne;
import jakarta.persistence.PrePersist;
import jakarta.persistence.Table;

@Entity
@Table(name = "contests")
public class Contest {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Long id;

    @Column(nullable = false, length = 200)
    private String name;

    @Column(columnDefinition = "text")
    private String description;

    @Column(name = "start_time", nullable = false)
    private Instant startTime;

    @Column(name = "end_time", nullable = false)
    private Instant endTime;

    @ManyToOne(fetch = FetchType.LAZY)
    @JoinColumn(name = "created_by", updatable = false)
    private User createdBy;

    @Column(name = "created_at", nullable = false, updatable = false)
    private Instant createdAt;

    @Column(name = "is_private", nullable = false)
    private boolean isPrivate;

    @Column(name = "freeze_minutes", nullable = false)
    private int freezeMinutes = 60;

    @Column(name = "problems_published_at")
    private Instant problemsPublishedAt;

    protected Contest() {
    }

    public Contest(String name, String description, Instant startTime, Instant endTime,
                   boolean isPrivate, int freezeMinutes, User createdBy) {
        this.name = name;
        this.description = description;
        this.startTime = startTime;
        this.endTime = endTime;
        this.isPrivate = isPrivate;
        this.freezeMinutes = freezeMinutes;
        this.createdBy = createdBy;
    }

    @PrePersist
    void onCreate() {
        this.createdAt = Instant.now();
    }

    /** Metadata edits are guarded to before-start by the service layer (PRD §7.8). */
    public void updateMetadata(String name, String description, Instant startTime, Instant endTime,
                               boolean isPrivate, int freezeMinutes) {
        this.name = name;
        this.description = description;
        this.startTime = startTime;
        this.endTime = endTime;
        this.isPrivate = isPrivate;
        this.freezeMinutes = freezeMinutes;
    }

    /** Set by the auto-publish job; the marker makes publishing exactly-once per contest. */
    public void markProblemsPublished(Instant publishedAt) {
        this.problemsPublishedAt = publishedAt;
    }

    public Long getId() {
        return id;
    }

    public String getName() {
        return name;
    }

    public String getDescription() {
        return description;
    }

    public Instant getStartTime() {
        return startTime;
    }

    public Instant getEndTime() {
        return endTime;
    }

    public User getCreatedBy() {
        return createdBy;
    }

    public Instant getCreatedAt() {
        return createdAt;
    }

    public boolean isPrivate() {
        return isPrivate;
    }

    public int getFreezeMinutes() {
        return freezeMinutes;
    }

    public Instant getProblemsPublishedAt() {
        return problemsPublishedAt;
    }
```

```java
package com.onlinejudge.backend.repository;

import java.time.Instant;
import java.util.Locale;

import org.springframework.data.jpa.domain.Specification;

import com.onlinejudge.common.entity.Contest;

/** Contest list filters (search + derived status windows). */
public final class ContestSpecifications {

    private ContestSpecifications() {
    }

    public static Specification<Contest> nameContains(String search) {
        String pattern = "%" + search.toLowerCase(Locale.ROOT)
                .replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_") + "%";
        return (root, query, cb) -> cb.like(cb.lower(root.get("name")), pattern, '\\');
    }

    public static Specification<Contest> isUpcoming(Instant now) {
        return (root, query, cb) -> cb.greaterThan(root.get("startTime"), now);
    }

    public static Specification<Contest> isRunning(Instant now) {
        return (root, query, cb) -> cb.and(
                cb.lessThanOrEqualTo(root.get("startTime"), now),
                cb.greaterThan(root.get("endTime"), now));
    }

    public static Specification<Contest> isEnded(Instant now) {
        return (root, query, cb) -> cb.lessThanOrEqualTo(root.get("endTime"), now);
    }

    /** Public listing is always restricted to non-private contests. */
    public static Specification<Contest> publicOnly() {
        return (root, query, cb) -> cb.isFalse(root.get("isPrivate"));
    }
}
```

```java
        Specification<Contest> spec = ContestSpecifications.publicOnly();
        if (status != null) {
            spec = spec.and(switch (status) {
                case UPCOMING -> ContestSpecifications.isUpcoming(now);
                case RUNNING -> ContestSpecifications.isRunning(now);
                case ENDED -> ContestSpecifications.isEnded(now);
            });

        Contest contest = visibleContest(id, usernameOrNull);
        ContestStatus status = ContestStatus.of(contest.getStartTime(), contest.getEndTime(), now);
        User caller = callerOrNull(usernameOrNull);
        boolean isAdmin = isAdmin(caller);
        boolean registered = caller != null
                && contestParticipantRepository.existsByContestIdAndUserId(id, caller.getId());
        boolean problemsRevealed = isAdmin || status != ContestStatus.UPCOMING;
        List<ContestProblemSummaryResponse> problems = problemsRevealed ? problems(id, caller) : null;
        return new ContestDetailResponse(contest.getId(), contest.getName(), contest.getDescription(),
                contest.getStartTime(), contest.getEndTime(), status, contest.getFreezeMinutes(),
                contestParticipantRepository.countByContestId(id), registered,
                !contest.isPrivate() && status == ContestStatus.UPCOMING, problemsRevealed, problems);
    }

    @Override

    @PreAuthorize("isAuthenticated()")
    public RegistrationResponse register(Long id, String username) {
        Contest contest = visibleContest(id, username);
        if (contest.isPrivate()) {
            throw new ResourceNotFoundException("Contest not found: " + id);

        requireUpcoming(contest);
        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new ResourceNotFoundException("User does not exist"));
        if (!contestParticipantRepository.existsByContestIdAndUserId(id, user.getId())) {
            contestParticipantRepository.save(new ContestParticipant(contest, user));
        }
        return new RegistrationResponse(true);
    }

    @Override
    @Transactional
    @PreAuthorize("isAuthenticated()")

        User user = userRepository.findByUsername(username)
                .orElseThrow(() -> new ResourceNotFoundException("User does not exist"));
        contestParticipantRepository.findByContestId(id).stream()
                .filter(participant -> participant.getUser().getId().equals(user.getId()))
                .findFirst()

            throw new ResourceNotFoundException("Contest not found: " + id);

                .anyMatch(role -> AppRole.ROLE_ADMIN.name().equals(role.getName()));

        Set<Long> solved = Set.of();

            solved = new HashSet<>(submissionRepository.findProblemIdsSolvedBy(

        Set<Long> solvedIds = solved;
        return links.stream().map(link -> new ContestProblemSummaryResponse(link.getProblem().getId(),
                link.getProblem().getSlug(), link.getProblem().getTitle(), link.getProblem().getDifficulty(),
                link.getDisplayOrder(), link.getPoints(),
                solvedIds.contains(link.getProblem().getId()))).toList();
    }

        return size <= 0 ? DEFAULT_PAGE_SIZE : Math.min(size, MAX_PAGE_SIZE);
```

```java
package com.onlinejudge.backend.service;

/**
 * Publishes the problems of one ended public contest. Called by the scheduler; a separate
 * bean so each contest gets its own transaction (one bad contest cannot block the rest)
 * and so tests can invoke it directly (PRD §7.8).
 */
@Service
public class ContestPublicationService {

    private static final Logger log = LoggerFactory.getLogger(ContestPublicationService.class);

    private final ContestRepository contestRepository;
    private final ContestProblemRepository contestProblemRepository;
    private final TestCaseRepository testCaseRepository;
    private final ProblemRepository problemRepository;

    // constructor injection ...

    @Transactional
    public void publishContest(Long contestId) {
        Contest contest = contestRepository.findById(contestId).orElse(null);
        if (contest == null || contest.isPrivate() || contest.getProblemsPublishedAt() != null) {
            return;
        }
        if (contest.getEndTime().isAfter(Instant.now())) {
            return;
        }
        for (ContestProblem link : contestProblemRepository.findByContestIdOrderByDisplayOrderAsc(contestId)) {
            Problem problem = link.getProblem();
            if (problem.isPublished()) {
                continue;
            }
            if (testCaseRepository.countByProblemId(problem.getId()) == 0) {
                log.warn("Contest {} ended but problem {} has no test cases; leaving unpublished",
                        contestId, problem.getId());
                continue;
            }
            problem.setPublished(true);
            problemRepository.save(problem);
            log.info("Published contest problem {} after contest {} ended", problem.getId(), contestId);
        }
        contest.markProblemsPublished(Instant.now());
        contestRepository.save(contest);
    }
}
```

```java
@Service
@Transactional(readOnly = true)
public class StandingsServiceImpl implements StandingsService {

    private static final int WRONG_ATTEMPT_PENALTY_MINUTES = 10;

    private final ContestRepository contestRepository;
    private final ContestProblemRepository contestProblemRepository;
    private final ContestParticipantRepository contestParticipantRepository;
    private final SubmissionRepository submissionRepository;
    private final UserRepository userRepository;
    private final StandingsCache standingsCache;

    // constructor injection ...

    @Override
    public StandingsResponse standings(Long contestId, String usernameOrNull) {
        Contest contest = contestRepository.findById(contestId)
                .orElseThrow(() -> new ResourceNotFoundException("Contest not found: " + contestId));
        User caller = usernameOrNull == null ? null
                : userRepository.findByUsername(usernameOrNull).orElse(null);
        boolean admin = caller != null && caller.getRoles().stream()
                .anyMatch(role -> AppRole.ROLE_ADMIN.name().equals(role.getName()));
        if (contest.isPrivate() && !admin) {
            throw new ResourceNotFoundException("Contest not found: " + contestId);
        }
        Optional<StandingsResponse> cached = standingsCache.get(contestId);
        if (cached.isPresent()) {
            return cached.get();
        }
        StandingsResponse computed = compute(contest);
        standingsCache.put(contestId, computed);
        return computed;
    }

    private StandingsResponse compute(Contest contest) {
        Instant now = Instant.now();
        ContestStatus status = ContestStatus.of(contest.getStartTime(), contest.getEndTime(), now);
        boolean freezeActive = contest.getFreezeMinutes() > 0 && status == ContestStatus.RUNNING
                && !now.isBefore(contest.getEndTime().minusSeconds(contest.getFreezeMinutes() * 60L));
        Instant frozenAt = freezeActive
                ? contest.getEndTime().minusSeconds(contest.getFreezeMinutes() * 60L)
                : null;
        Instant cutoff = freezeActive ? frozenAt : contest.getEndTime();

        List<ContestProblem> links = contestProblemRepository.findByContestIdOrderByDisplayOrderAsc(contest.getId());
        List<ContestParticipant> participants = contestParticipantRepository.findByContestId(contest.getId());
        if (links.isEmpty() || participants.isEmpty()) {
            return new StandingsResponse(freezeActive, frozenAt, List.of());
        }
        List<Long> problemIds = links.stream().map(link -> link.getProblem().getId()).toList();
        Map<Long, Integer> pointsByProblem = links.stream().collect(Collectors.toMap(
                link -> link.getProblem().getId(), ContestProblem::getPoints));
        Map<Long, String> usernameByUser = participants.stream().collect(Collectors.toMap(
                participant -> participant.getUser().getId(),
                participant -> participant.getUser().getUsername()));
        List<Long> userIds = List.copyOf(usernameByUser.keySet());

        List<Submission> submissions = submissionRepository.findForStandings(
                problemIds, userIds, contest.getStartTime(), cutoff);
        Map<Long, List<Submission>> byUserAndProblem = submissions.stream().collect(Collectors.groupingBy(
                submission -> key(submission.getUser().getId(), submission.getProblem().getId())));

        List<Row> rows = new ArrayList<>();
        for (Long userId : userIds) {
            int totalPoints = 0;
            int penalty = 0;
            Instant lastAc = null;
            List<StandingsProblemCell> cells = new ArrayList<>();
            for (Long problemId : problemIds) {
                List<Submission> mine = byUserAndProblem.getOrDefault(key(userId, problemId), List.of()).stream()
                        .sorted(Comparator.comparing(Submission::getSubmittedAt))
                        .toList();
                Submission firstAc = mine.stream()
                        .filter(submission -> submission.getVerdict() == Verdict.ACCEPTED)
                        .findFirst().orElse(null);
                int wrongAttempts = 0;
                for (Submission submission : mine) {
                    if (firstAc != null && !submission.getSubmittedAt().isBefore(firstAc.getSubmittedAt())) {
                        break;
                    }
                    if (submission.getVerdict() != Verdict.ACCEPTED
                            && submission.getVerdict() != Verdict.SYSTEM_ERROR) {
                        wrongAttempts++;
                    }
                }
                if (firstAc != null) {
                    int minute = (int) Duration.between(contest.getStartTime(),
                            firstAc.getSubmittedAt()).toMinutes();
                    totalPoints += pointsByProblem.getOrDefault(problemId, 0);
                    penalty += minute + WRONG_ATTEMPT_PENALTY_MINUTES * wrongAttempts;
                    lastAc = lastAc == null || firstAc.getSubmittedAt().isAfter(lastAc)
                            ? firstAc.getSubmittedAt() : lastAc;
                    cells.add(new StandingsProblemCell(problemId, true, wrongAttempts, minute));
                } else {
                    cells.add(new StandingsProblemCell(problemId, false, wrongAttempts, null));
                }
            }
            rows.add(new Row(userId, usernameByUser.get(userId), totalPoints, penalty, lastAc, cells));
        }
        rows.sort(Comparator.comparingInt(Row::totalPoints).reversed()
                .thenComparingInt(Row::penalty)
                .thenComparing(row -> row.lastAc, Comparator.nullsLast(Comparator.naturalOrder()))
                .thenComparing(row -> row.userId));
        List<StandingsRow> ranked = new ArrayList<>();
        for (int index = 0; index < rows.size(); index++) {
            Row row = rows.get(index);
            int rank = index + 1;
            if (index > 0) {
                Row previous = rows.get(index - 1);
                if (previous.totalPoints == row.totalPoints && previous.penalty == row.penalty) {
                    rank = ranked.get(index - 1).rank();
                }
            }
            ranked.add(new StandingsRow(rank, row.username, row.totalPoints, row.penalty, row.cells));
        }
        return new StandingsResponse(freezeActive, frozenAt, ranked);
    }

    private static long key(Long userId, Long problemId) {
        return userId * 1_000_000_007L + problemId;
    }

    private record Row(Long userId, String username, int totalPoints, int penalty, Instant lastAc,
                       List<StandingsProblemCell> cells) {
    }
}
```

```ts
  solvedByMe: boolean

  id: number
  name: string
  description: string | null
  startTime: string
  endTime: string
  status: ContestStatus
  freezeMinutes: number
  participantCount: number
  registered: boolean
  registrationOpen: boolean
  problemsRevealed: boolean

  problemId: number
  slug: string
  title: string
  statement: string
  difficulty: Difficulty
  timeLimitMs: number
  memoryLimitKb: number
  points: number
  contestId: number
  contestName: string

  problemId: number
  solved: boolean
  wrongAttempts: number
  acceptedAtMinute: number | null
}

export interface StandingsRow {
  rank: number
  username: string
  totalPoints: number
  penalty: number
  problems: StandingsProblemCell[]

  frozen: boolean
  frozenAt: string | null
  rows: StandingsRow[]
}

export interface ContestFilters {
  status?: ContestStatus | ''
  page?: number
  size?: number
}

export function fetchContests(filters: ContestFilters): Promise<PagedResponse<ContestSummary>> {
  const params = new URLSearchParams()
  if (filters.status) params.set('status', filters.status)
  if (filters.page !== undefined) params.set('page', String(filters.page))
  if (filters.size !== undefined) params.set('size', String(filters.size))
  return apiFetch<PagedResponse<ContestSummary>>(`/api/v1/contests?${params.toString()}`)
}

export function fetchContest(id: number): Promise<ContestDetail> {
  return apiFetch<ContestDetail>(`/api/v1/contests/${id}`)
}

export function registerForContest(id: number): Promise<{ registered: boolean }> {
  return apiFetch(`/api/v1/contests/${id}/register`, { method: 'POST' })
}

export function unregisterFromContest(id: number): Promise<void> {
  return apiFetch(`/api/v1/contests/${id}/register`, { method: 'DELETE' })
}

export function fetchContestProblem(id: number, slug: string): Promise<ContestProblemDetail> {

    `/api/v1/contests/${id}/problems/${encodeURIComponent(slug)}`,
  )
}

export function fetchStandings(id: number): Promise<StandingsResponse> {
```

```tsx
import { useNavigate, useSearchParams } from 'react-router-dom'
import { useQuery } from '@tanstack/react-query'
import { fetchContests } from '../api/contests'
import type { ContestStatus } from '../api/contests'
import { ContestStatusChip } from '../components/ContestStatusChip'
import { ErrorBanner } from '../components/ErrorBanner'
import { Pagination } from '../components/Pagination'
import { Spinner } from '../components/Spinner'
import { toIntParam } from '../helpers/queryParams'

const TABS: { label: string; value: ContestStatus | '' }[] = [
  { label: 'All', value: '' },
  { label: 'Upcoming', value: 'UPCOMING' },
  { label: 'Running', value: 'RUNNING' },
  { label: 'Past', value: 'ENDED' },
]

export function ContestsPage() {
  const [params, setParams] = useSearchParams()
  const navigate = useNavigate()
  const status = (params.get('status') as ContestStatus | null) ?? ''
  const page = toIntParam(params.get('page'), 0)

  const query = useQuery({
    queryKey: ['contests', { status, page }],
    queryFn: () => fetchContests({ status, page, size: 20 }),
  })

  function setParam(patch: { status?: string; page?: number }) {
    const next = new URLSearchParams(params)
    if (patch.status !== undefined) {
      if (patch.status === '') next.delete('status')
      else next.set('status', patch.status)
      next.delete('page')
    }
    if (patch.page !== undefined) next.set('page', String(patch.page))
    setParams(next)
  }

  return (
    <section>
      <h1>Contests</h1>
      <div className="contest-tabs">
        {TABS.map((tab) => (
          <button
            key={tab.label}
            type="button"
            className={status === tab.value ? 'btn btn--solid' : 'btn btn--outline'}
            onClick={() => setParam({ status: tab.value })}
          >
            {tab.label}
          </button>
        ))}
      </div>
      {query.isPending && <Spinner label="Loading contests" />}
      {query.isError && <ErrorBanner error={query.error} />}
      {query.isSuccess && query.data.content.length === 0 && <p>No contests found.</p>}
      {query.isSuccess && query.data.content.length > 0 && (
        <>
          <table className="problem-table">
            <thead>
              <tr>
                <th>Name</th>
                <th>Status</th>
                <th>Start</th>
                <th>End</th>
                <th>Participants</th>
              </tr>
            </thead>
            <tbody>
              {query.data.content.map((contest) => (
                <tr key={contest.id}>
                  <td>
                    <button
                      type="button"
                      className="linklike"
                      onClick={() => navigate(`/contests/${contest.id}`)}
                    >
                      {contest.name}
                    </button>
                  </td>
                  <td>
                    <ContestStatusChip status={contest.status} />
                  </td>
                  <td>{new Date(contest.startTime).toLocaleString()}</td>
                  <td>{new Date(contest.endTime).toLocaleString()}</td>
                  <td>{contest.participantCount}</td>
                </tr>
              ))}
            </tbody>
          </table>
          <Pagination
            page={page}
            totalPages={query.data.totalPages}
            onPageChange={(next) => setParam({ page: next })}
          />
        </>
      )}
    </section>
  )
}
```

```tsx
import { useEffect, useState } from 'react'

function formatRemaining(ms: number): string {
  const total = Math.max(0, Math.floor(ms / 1000))
  const days = Math.floor(total / 86400)
  const hours = Math.floor((total % 86400) / 3600)
  const minutes = Math.floor((total % 3600) / 60)
  const seconds = total % 60
  const pad = (value: number) => String(value).padStart(2, '0')
  return days > 0
    ? `${days}d ${pad(hours)}:${pad(minutes)}:${pad(seconds)}`
    : `${pad(hours)}:${pad(minutes)}:${pad(seconds)}`
}

export function Countdown({ target, label }: { target: string; label: string }) {
  const [remaining, setRemaining] = useState(() => Date.parse(target) - Date.now())

  useEffect(() => {
    const timer = window.setInterval(() => setRemaining(Date.parse(target) - Date.now()), 1000)
    return () => window.clearInterval(timer)
  }, [target])

  return (
    <span className="contest-countdown">
      {label}: {formatRemaining(remaining)}
    </span>
  )
}
```

```ts
export interface AdminContestSummary {
  id: number
  name: string
  startTime: string
  endTime: string
  status: 'UPCOMING' | 'RUNNING' | 'ENDED'
  isPrivate: boolean
  participantCount: number
  problemCount: number
  problemsPublishedAt: string | null
}

export interface CreateContestBody {
  name: string
  description?: string
  startTime: string
  endTime: string
  isPrivate: boolean
  freezeMinutes?: number
}

export type UpdateContestBody = CreateContestBody

export interface AttachableProblem {
  id: number
  slug: string
  title: string
  difficulty: Difficulty
}

export interface ContestProblemCreated {
  problemId: number
  slug: string
}

export function listAdminContests(filters: {
  search?: string
  status?: string
  page?: number
  size?: number
}): Promise<PagedResponse<AdminContestSummary>> {
  const params = new URLSearchParams()
  if (filters.search) params.set('search', filters.search)
  if (filters.status) params.set('status', filters.status)
  if (filters.page !== undefined) params.set('page', String(filters.page))
  if (filters.size !== undefined) params.set('size', String(filters.size))
  return apiFetch(`/api/v1/admin/contests?${params.toString()}`)
}

export function createContest(body: CreateContestBody): Promise<{ id: number; name: string }> {
  return apiFetch('/api/v1/admin/contests', { method: 'POST', body: JSON.stringify(body) })
}

export function updateContest(id: number, body: UpdateContestBody): Promise<void> {
  return apiFetch(`/api/v1/admin/contests/${id}`, { method: 'PUT', body: JSON.stringify(body) })
}

export function deleteContest(id: number): Promise<void> {
  return apiFetch(`/api/v1/admin/contests/${id}`, { method: 'DELETE' })
}

export function fetchAttachableProblems(search?: string): Promise<AttachableProblem[]> {
  const params = new URLSearchParams()
  if (search) params.set('search', search)
  return apiFetch(`/api/v1/admin/contests/attachable-problems?${params.toString()}`)
}

export function attachContestProblem(
  contestId: number,
  body: { problemId: number; points?: number },
): Promise<void> {
  return apiFetch(`/api/v1/admin/contests/${contestId}/problems`, {
    method: 'POST',
    body: JSON.stringify(body),
  })
}

export function createContestProblemInline(
  contestId: number,
  body: {
    title: string
    statement: string
    difficulty: Difficulty
    timeLimitMs?: number
    memoryLimitKb?: number
    points?: number
  },
): Promise<ContestProblemCreated> {
  return apiFetch(`/api/v1/admin/contests/${contestId}/problems/new`, {
    method: 'POST',
    body: JSON.stringify(body),
  })
}

export function updateContestProblem(
  contestId: number,
  problemId: number,
  body: { points?: number; displayOrder?: number },
): Promise<void> {
  return apiFetch(`/api/v1/admin/contests/${contestId}/problems/${problemId}`, {
    method: 'PUT',
    body: JSON.stringify(body),
  })
}

export function detachContestProblem(contestId: number, problemId: number): Promise<void> {
  return apiFetch(`/api/v1/admin/contests/${contestId}/problems/${problemId}`, {
    method: 'DELETE',
  })
}
```
